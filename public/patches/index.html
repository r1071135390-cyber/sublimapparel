One-shot apply script (recommended)

Save this as apply-patches.sh in your sublimapparel repo root and run bash apply-patches.sh:

#!/bin/bash
set -e
mkdir -p .patches
cd .patches
BASE="https://279a63c8-c31c-4ac6-a941-1577a843d2e1.dev.coze.site/patches"
FILES=(
  "0001-feat-100-all-over-print.patch"
  "0002-feat.patch"
  "0003-feat.patch"
  "0004-feat.patch"
  "0005-feat.patch"
  "0006-feat.patch"
  "0007-feat.patch"
  "0008-feat.patch"
  "0009-feat.patch"
  "0010-feat.patch"
)
for f in "${FILES[@]}"; do
  if [ ! -f "$f" ]; then
    echo "↓ Downloading $f"
    curl -fsSL -O "$BASE/$f"
  fi
done
cd ..
git am --abort 2>/dev/null || true
for f in "${FILES[@]}"; do
  echo "✓ Applying $f"
  git am --signoff < ".patches/$f"
done
echo ""
echo "Done! All 10 patches applied. Now run:"
echo "  git push origin main"
