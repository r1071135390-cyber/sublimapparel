4. The Categorization Algorithm

Used to redistribute the 38 Sportswear products (see 07-sportswear-reclassification.tsv):

function categorizeByName(name) {
  const n = name.toLowerCase();

  // Rule 1: Bottom-wear → Pants
  if (n.includes("shorts") || n.includes("legging") ||
      n.includes("tights") ||
      (n.includes("swimwear") && n.includes("trunks"))) return "Pants";

  // Rule 2: Tank-top dress → Tank Top & Camis
  if (n.includes("vest") && n.includes("tank") && n.includes("dress")) {
    return "Tank Top & Camis"; // e.g. 0008
  }

  // Rule 3: Other vests → Vest
  if (n.includes("vest")) return "Vest";

  // Rule 4: Tight-fitting tops → Tank Top & Camis
  if (n.includes("rash guard") || n.includes("lycra top") ||
      n.includes("lycra suit")) return "Tank Top & Camis";

  // Rule 5: Singlet → T-Shirt
  if (n.includes("singlet")) return "T-Shirt";

  // Rule 6: Dress / leotard → Skirt
  if (n.includes("leotard")) return "Skirt";
  if (n.includes("dress") && !n.includes("tank vest")) return "Skirt";

  // Rule 7: Uniform with skirt → Skirt
  if (n.includes("uniform") &&
      (n.includes("skirt") || n.includes("top + skirt"))) return "Skirt";

  // Rule 8: Other sport tops → T-Shirt
  if (n.includes("jersey") || n.includes("outfit") ||
      n.includes("wear") || n.includes("set") ||
      n.includes("suit") || n.includes("gi") ||
      n.includes("fishing shirt")) return "T-Shirt";

  // Fallback: T-Shirt
  return "T-Shirt";
}
